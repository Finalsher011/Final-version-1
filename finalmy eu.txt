        TextColor3 = WHITE,
        Font = Enum.Font.Gotham,
        TextSize = 12,

    }, button)

    local indicator = Create("Frame", {
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -13, 0.5, 0),
        Size = UDim2.fromOffset(35, 18),
        BackgroundColor3 = Color3.fromRGB(55, 55, 60)
    }, button)

    Create("UICorner", {CornerRadius = UDim.new(1, 0)}, indicator)

    local function Refresh()
        local enabled = Config[key]
        indicator.BackgroundColor3 = enabled and RED or Color3.fromRGB(55, 55, 60)
    end

    button.MouseButton1Click:Connect(function()
        Config[key] = not Config[key]
        Refresh()
    end)

    Refresh()
    return button
end

local function Slider(parent, label, key, min, max)
    local holder = Create("Frame", {
        Size = UDim2.new(1, -8, 0, 58),
        BackgroundColor3 = PANEL
    }, parent)

    Create("UICorner", {CornerRadius = UDim.new(0, 8)}, holder)

    local valueLabel = Create("TextLabel", {
        Position = UDim2.fromOffset(12, 5),
        Size = UDim2.new(1, -24, 0, 22),
        BackgroundTransparency = 1,
        TextColor3 = WHITE,
        Font = Enum.Font.Gotham,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left
    }, holder)

    local track = Create("Frame", {
        Position = UDim2.fromOffset(12, 36),
        Size = UDim2.new(1, -24, 0, 5),
        BackgroundColor3 = Color3.fromRGB(55, 55, 60)
    }, holder)

    local fill = Create("Frame", {
        Size = UDim2.fromScale(0, 1),
        BackgroundColor3 = RED
    }, track)

    local dragging = false

    local function SetValue(x)
        local alpha = math.clamp(
            (x - track.AbsolutePosition.X) / track.AbsoluteSize.X,
            0, 1
        )
        Config[key] = math.floor(min + (max - min) * alpha)
        fill.Size = UDim2.fromScale(alpha, 1)
        valueLabel.Text = label .. ": " .. Config[key]
    end

    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            SetValue(input.Position.X)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            SetValue(input.Position.X)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

    SetValue(track.AbsolutePosition.X + track.AbsoluteSize.X *
        ((Config[key] - min) / (max - min)))

    return holder
end

-- NPCs: coloque modelos de treino em workspace.TrainingNPCs
local NPCFolder = workspace:FindFirstChild("TrainingNPCs")

local function IsValidNPC(model)
    if not model:IsA("Model") then return false end
    if Players:GetPlayerFromCharacter(model) then return false end

    local humanoid = model:FindFirstChildOfClass("Humanoid")
    local root = model:FindFirstChild("HumanoidRootPart")

    return humanoid and humanoid.Health > 0 and root
end

local function GetNPCs()
    local folder = workspace:FindFirstChild("TrainingNPCs")
    if not folder then return {} end

    local result = {}
    for _, model in ipairs(folder:GetChildren()) do
        if IsValidNPC(model) then
            table.insert(result, model)
        end
    end
    return result
end

-- ESP centralizado
local function UpdateESP()
    local current = GetNPCs()
    local seen = {}

    for _, npc in ipairs(current) do
        seen[npc] = true

        if Config.ESP then
            local highlight = Highlights[npc]

            if not highlight then
                highlight = Instance.new("Highlight")
                highlight.FillColor = RED
                highlight.OutlineColor = WHITE
                highlight.FillTransparency = 0.65
                highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                highlight.Parent = npc
                Highlights[npc] = highlight
            end

            highlight.Enabled = true
        elseif Highlights[npc] then
            Highlights[npc].Enabled = false
        end
    end

    for npc, highlight in pairs(Highlights) do
        if not seen[npc] or not npc.Parent then
            highlight:Destroy()
            Highlights[npc] = nil
        end
    end
end

-- Aimbot de treino: seleciona NPC mais próximo do centro da tela
local function GetBestTarget()
    local bestPart
    local bestDistance = Config.AimFOV

    for _, npc in ipairs(GetNPCs()) do
        local part = npc:FindFirstChild(Config.AimPart)
            or npc:FindFirstChild("Head")
            or npc:FindFirstChild("HumanoidRootPart")

        if part then
            local worldDistance = (part.Position - Camera.CFrame.Position).Magnitude

            if worldDistance <= Config.MaxDistance then
                local screenPos, visible = Camera:WorldToViewportPoint(part.Position)

                if visible then
                    local center = Camera.ViewportSize / 2
                    local screenDistance = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude

                    if screenDistance < bestDistance then
                        bestDistance = screenDistance
                        bestPart = part
                    end
                end
            end
        end
    end

    return bestPart
end

local elapsed = 0

Track(RunService.RenderStepped:Connect(function(dt)
    elapsed += dt

    if elapsed >= 0.08 then
        elapsed = 0
        UpdateESP()
    end

    Camera.FieldOfView = Config.FOV

    if Config.Fullbright then
        Lighting.Brightness = 3
        Lighting.ClockTime = 14
        Lighting.GlobalShadows = false
    end

    if Config.Aimbot and UserInputService:IsMouseButtonPressed(
        Enum.UserInputType.MouseButton2
    ) then
        local target = GetBestTarget()

        if target then
            local goal = CFrame.lookAt(
                Camera.CFrame.Position,
                target.Position
            )

            Camera.CFrame = Camera.CFrame:Lerp(
                goal,
                math.clamp(Config.AimSmoothness, 0.01, 1)
            )
        end
    end
end))

    end
end)

    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)
